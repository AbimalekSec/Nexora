function filterPid() {
    const allPids = processIDs('*');
    const fivemPid = processIDs("FiveM_*")[0];

    return allPids.filter(pid => 
        !SYSTEM_PIDS_TO_IGNORE.includes(pid) &&
        pid !== fivemPid
    );
}

function shouldScanProcess(processInfo) {
    if (!processInfo) return false;
    const lowerName = processInfo.name.toLowerCase();
    return !FIVEM_PROCESS_NAMES.some(name => lowerName.includes(name.toLowerCase()));
}

async function scanMemory(pid) {
    const info = processInfo(pid);
    if (!shouldScanProcess(info)) return;

    const foundStage1 = await strings([pid], CHEAT_STRINGS_STAGE_1, true);
    const foundStage2 = await strings([pid], CHEAT_STRINGS_STAGE_2, true);

    const selfHostNames = ["medalencoder", "vgtray", "discord", "obs64", "nvcontainer"];
    const infoNameLower = info.name.toLowerCase();

    if (selfHostNames.some(n => infoNameLower.includes(n))) {
        const requiredHooks = ["imgui_impl_dx11", "imgui_impl_win32"];
        const hasImGui = requiredHooks.every(str => foundStage1.includes(str));
        
        if (hasImGui) {
            const extra = foundStage1.filter(str => !requiredHooks.includes(str));
            if (extra.length > 0) {
                result(`Injected Cheat Menu in Overlay Host [[${info.name}]]`, severe);
                log(`[MEM][HOOK] Found in ${info.name} (${pid}): ${extra.join(", ")}`);
            }
        }
    } else if (foundStage1?.length) {
        result(`FiveM Cheat Artifacts in Memory [[${info.name}]]`, severe);
        log(`[MEM][STAGE1] Found in ${info.name} (${pid}): ${foundStage1.join(", ")}`);
    }

    if (foundStage2?.length) {
        result(`Prohibited FiveM Cheat Signatures [[${info.name}]]`, severe);
        log(`[MEM][STAGE2] Found in ${info.name} (${pid}): ${foundStage2.join(", ")}`);
    }
}

async function runMemoryScan() {
    const pids = filterPid();
    const maxConcurrency = 16;
    for (let i = 0; i < pids.length; i += maxConcurrency) {
        const batch = pids.slice(i, i + maxConcurrency);
        await Promise.all(batch.map(pid => scanMemory(pid)));
    }
}
